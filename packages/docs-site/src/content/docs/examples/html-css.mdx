## Blueprint

```js
{
  title: "HTML + CSS page",
  variables: [{ key: "name", prompt: "Page name" }],
  snippets: {
    page: [
      "<!DOCTYPE html>",
      '<html lang="en">',
      "  <head>",
      '    <meta charset="UTF-8" />',
      '    <meta name="viewport" content="width=device-width, initial-scale=1.0" />',
      "    <title>{name.raw}</title>",
      '    <link rel="stylesheet" href="style.css" />',
      "  </head>",
      "  <body>",
      '    <main class="{name.kebab}">',
      "      <h1>{name.raw}</h1>",
      "    </main>",
      "  </body>",
      "</html>",
    ],
    style: [
      "html {",
      "  box-sizing: border-box;",
      "}",
      "",
      "*,",
      "*::before,",
      "*::after {",
      "  box-sizing: inherit;",
      "  padding: 0;",
      "  margin: 0;",
      "}",
      "",
      "body {",
      "  font-family: sans-serif;",
      "}",
      "",
      ".{name.kebab} {",
      "  max-width: 40rem;",
      "  margin: 0 auto;",
      "  padding: 2rem;",
      "}",
    ],
  },
  structure: {
    folders: [
      {
        name: "{name.kebab}",
        files: [
          { name: "index.html", snippet: "page" },
          { name: "style.css", snippet: "style" },
        ],
      },
    ],
  },
}
```
